  process.env.SUPABASE_URL,
  process.env.SUPABASE_KEY
)

// Gmail Auth
const auth = new google.auth.GoogleAuth({
  keyFile: 'credentials.json',
  scopes: ['https://www.googleapis.com/auth/gmail.readonly']
})

// This endpoint Google calls when new email arrives
app.post('/webhook/gmail', async (req, res) => {

  // 1. Acknowledge Google immediately
  res.sendStatus(200)

  // 2. Decode the notification
  const data = req.body.message?.data
  if (!data) return
  
  // 3. Fetch latest email from Gmail
  const gmail = google.gmail({ version: 'v1', auth })
  const messages = await gmail.users.messages.list({
    userId: 'me',
    q: 'from:IndusInd is:unread',
    maxResults: 1
  })

  const msg = await gmail.users.messages.get({
    userId: 'me',
    id: messages.data.messages[0].id,
    format: 'full'
  })

  // 4. Extract email body
  const body = Buffer.from(
    msg.data.payload.parts?.[0]?.body?.data || 
    msg.data.payload.body?.data, 
    'base64'
  ).toString()

  // 5. Extract UTR and Amount
  const utr = body.match(/UTR[:\s]+([A-Z0-9]+)/i)?.[1]

  await supabase
    .from('orders')
    .update({ 
      status: 'paid', 
      utr: utr

export default {
  async email(message, env, ctx) {

    // 1. Read the email
    const reader = message.raw.getReader()
    let chunks = []
    
    while (true) {
      const { done, value } = await reader.read()
      if (done) break
      chunks.push(value)
    }
    
    const raw = new TextDecoder().decode(
      Uint8Array.from(chunks.flatMap(c => [...c]))
    )

    // 2. Check it's from IndusInd
    const from = message.from.toLowerCase()
    if (!from.includes('indusind')) {
      message.forward('your-personal@gmail.com')
      return
    }

    // 3. Extract UTR and Amount
    const utr = raw.match(/UTR\s*(?:No\.?|Number)?[\s:]*([A-Z0-9]{12,22})/i)?.[1]
    const amount = raw.match(/Rs\.?\s*([\d,]+(?:\.\d{2})?)/i)?.[1]?.replace(/,/g, '')

    if (!utr || !amount) return

    // 4. Update Supabase
    const response = await fetch(
      `${env.SUPABASE_URL}/rest/v1/orders`,
      {
        method: 'PATCH',
        headers: {
          'apikey': env.SUPABASE_KEY,
          'Authorization': `Bearer ${env.SUPABASE_KEY}`,
          'Content-Type': 'application/json',
          'Prefer': 'return=minimal'
        },
        body: JSON.stringify({
          status: 'paid',
          utr: utr,
          paid_at: new Date().toISOString()
        })
      }
    )

    console.log(`✅ Payment verified: UTR ${utr} ₹${amount}`)
  }

export default {
  async email(message, env, ctx) {

    // 1. Only process IndusInd emails
    if (!message.from.toLowerCase().includes('indusind_bank@indusind.com')) {
      return
    }

    // 2. Read raw email
    const reader = message.raw.getReader()
    let chunks = []
    while (true) {
      const { done, value } = await reader.read()
      if (done) break
      chunks.push(value)
    }
    const raw = new TextDecoder().decode(
      Uint8Array.from(chunks.flatMap(c => [...c]))
    )

    // 3. Only process CREDIT transactions
    if (!raw.includes('Credited')) return

    // 4. Extract Amount → "INR 2.00"
    const amount = raw.match(/Credited\s+for\s+INR\s+([\d.]+)/i)?.[1]

    // 5. Extract UTR → "UPI/206754068147/CR/..."
    const utr = raw.match(/UPI\/(\d{12,})\//i)?.[1]

    if (!utr || !amount) {
      console.log('❌ Could not extract UTR or Amount')
      console.log('Raw snippet:', raw.substring(0, 500))
      return
    }

    console.log(`UTR: ${utr} | Amount: ₹${amount}`)

    // 6. Update Supabase — match by amount + pending status
    const { data, error } = await fetch(
      `${env.SUPABASE_URL}/rest/v1/orders?status=eq.pending&amount=eq.${amount}`,
      {
        method: 'PATCH',
        headers: {
          'apikey': env.SUPABASE_KEY,
          'Authorization': `Bearer ${env.SUPABASE_KEY}`,
          'Content-Type': 'application/json',
          'Prefer': 'return=representation'
        },
        body: JSON.stringify({
          status: 'paid',
          utr: utr,
          paid_at: new Date().toISOString()
        })
      }
    ).then(r => r.json())

    if (error) {
      console.log('❌ Supabase error:', error)
      return
    }

    console.log(`✅ Payment verified: UTR ${utr} ₹${amount}`)
  }
}
